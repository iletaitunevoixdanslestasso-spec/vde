1. DocumentTypeRepository
import { BaseRepository } from "./BaseRepository";


export class DocumentTypeRepository extends BaseRepository {


    constructor(table = "document_types") {
        super(table);
    }


    async findAllActive() {
        return this.supabase
            .from(this.table)
            .select("id, code, libelle")
            .is("deleted_at", null)
            .order("libelle", { ascending: true });
    }


    async findByCode(code) {
        return this.supabase
            .from(this.table)
            .select("*")
            .eq("code", code)
            .maybeSingle();
    }
}
2. Dans ReferentielDocumentService

Ajoute le repository DocumentTypeRepository :

import { BaseService } from "./BaseService";
import { DocumentTypeRepository } from "../repositories/DocumentTypeRepository";


export class ReferentielDocumentService extends BaseService {


    constructor(repository, validator = null, mapper = null) {
        super(repository, validator, mapper);


        this.documentTypeRepository = new DocumentTypeRepository("document_types");
    }


    async getAvailableDocumentTypes() {


        const { data, error } =
            await this.documentTypeRepository.findAllActive();


        if (error) {
            return {
                success: false,
                data: [],
                error: error.message
            };
        }


        return {
            success: true,
            data
        };
    }
}
